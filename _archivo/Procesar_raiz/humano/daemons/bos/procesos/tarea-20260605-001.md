```go
// recalcBodyHeight calcula m.bodyHeight según las líneas fijas de top/bottom.
// Debe llamarse en cada tea.WindowSizeMsg y en setScreen().
// topH: topbar(1)+stepper?(1)+title(1)+sep(1)+blank(1) → 5 con stepper, 4 sin
// botH: sep(1)+hint(1) = viewFooter() siempre 2 líneas
func (m *model) recalcBodyHeight() {
	topH := 4 // base: topbar + title + sep + blank (sin stepper)
	if m.showStepper {
		topH++ // +1 para el stepper del wizard
	}
	botH := 2 // viewFooter() siempre produce 2 líneas
	m.bodyHeight = m.height - topH - botH
	if m.bodyHeight < 1 {
		m.bodyHeight = 1
	}
}

// isInstallingScreen retorna true para P5/P5B/P5C que tienen sus propios viewports.
// Estas pantallas NO usan bodyVP global.
func (m model) isInstallingScreen() bool {
	return m.screen == ScreenInstalling ||
		m.screen == ScreenInstallLog ||
		m.screen == ScreenInstallErr
}

// setScreen cambia de pantalla actualizando showStepper, showNavHint y bodyHeight.
// TODOS los cambios de m.screen deben pasar por aquí.
func (m *model) setScreen(s Screen) {
	m.screen = s
	m.showStepper = (s >= ScreenWizardP1 && s <= ScreenInstalling)
	m.showNavHint = (s >= ScreenWizardP1 && s <= ScreenWizardP4)
	m.recalcBodyHeight()
}
```

```go
    toolbarH := 1

    if logVPH < 2 { logVPH = 2 }
    m.vpLog.Width  = m.width
    m.vpLog.Height = logVPH

    // Contenido: líneas de log con timestamp opcional
    sw := m.safeW()
    var logLines []string
    for _, e := range m.logs {
        logLines = append(logLines, e.render(sw-2))
    }
    if len(logLines) == 0 {
        logLines = []string{sDim.Render("Sin logs disponibles")}
    }
    m.vpLog.SetContent(strings.Join(logLines, "\n"))

    // Toolbar: título + hint de vuelta
    toolbar := sBold.Render("Log completo") + "  " +

}
```
